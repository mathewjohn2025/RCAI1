1.	Remove any useWhoAmI() call and any useEffect that auto-redirects based on whoami.
2.	Convert inputs to uncontrolled (no value= prop), add proper autocomplete, and memoize the component so parent re-renders don’t replace the inputs.
import React, { useRef, useEffect, useState } from "react";
import { useNavigate, useLocation } from "react-router-dom";
import { useQueryClient } from "@tanstack/react-query";
import { api } from "../api/http";

function AdminLoginInner() {
  const emailRef = useRef<HTMLInputElement>(null);
  const passRef = useRef<HTMLInputElement>(null);
  const [submitting, setSubmitting] = useState(false);
  const [err, setErr] = useState("");

  useEffect(() => { emailRef.current?.focus(); }, []); // focus once only

  const nav = useNavigate();
  const loc = useLocation();
  const qc = useQueryClient();

  function getReturnTo() {
    const params = new URLSearchParams(loc.search);
    const raw = params.get("returnTo");
    if (raw) {
      const dec = decodeURIComponent(raw);
      if (dec.startsWith("/admin")) return dec; // only allow admin paths
    }
    return "/admin/settings";
  }

  async function onSubmit(e: React.FormEvent) {
    e.preventDefault();
    setSubmitting(true); setErr("");
    try {
      const email = emailRef.current?.value || "";
      const password = passRef.current?.value || "";
      await api("/api/auth/login", {
        method: "POST",
        body: JSON.stringify({ email, password }),
      });

      // Wait briefly until whoami sees the session once
      await qc.invalidateQueries({ queryKey: ["whoami"] });
      const t0 = Date.now();
      while (Date.now() - t0 < 1500) {
        const me: any = qc.getQueryData(["whoami"]);
        if (me?.user) break;
        await qc.refetchQueries({ queryKey: ["whoami"], exact: true });
        await new Promise(r => setTimeout(r, 100));
      }
      nav(getReturnTo(), { replace: true });
    } catch (e: any) {
      setErr(e?.body?.error || "Login failed");
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <form onSubmit={onSubmit} noValidate>
      <label htmlFor="email">Email Address</label>
      <input
        ref={emailRef}
        id="email"
        name="username"           // correct for autofill
        type="email"
        inputMode="email"
        autoComplete="username"
        autoCapitalize="none"
        autoCorrect="off"
        spellCheck={false}
        defaultValue=""           // uncontrolled -> no value thrash
      />

      <label htmlFor="password">Password</label>
      <input
        ref={passRef}
        id="password"
        name="password"
        type="password"
        autoComplete="current-password"
        defaultValue=""
      />

      <button type="submit" disabled={submitting}>Sign In</button>
      {err && <p role="alert">{err}</p>}
    </form>
  );
}

export default React.memo(AdminLoginInner); // memoize to avoid replacement
