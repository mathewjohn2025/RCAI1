1.	Add a programmatic login API (same origin)
Create POST /api/auth/login that accepts { email, password }, verifies the admin user, sets the session, saves it, and responds 200.
// server/auth.ts
import type { Request, Response } from 'express';
import { findUserByEmail, verifyPassword } from './lib/users'; // existing helpers

export async function login(req: Request, res: Response) {
  const { email, password, returnTo } = req.body || {};
  const user = await findUserByEmail(email);
  if (!user) return res.status(401).json({ error: 'invalid' });
  const ok = await verifyPassword(password, user.passwordHash);
  if (!ok || user.role !== process.env.ADMIN_ROLE_NAME) {
    return res.status(401).json({ error: 'invalid' });
  }
  req.session.user = { id: user.id, role: user.role };
  req.session.save(() => res.status(200).json({ ok: true, role: user.role, returnTo: returnTo || '/admin' }));
}
Wire it up:
// server/index.ts
import express from 'express';
import session from 'express-session';
import { login } from './auth';

app.use(session({
  name: 'sid',
  secret: process.env.SESSION_SECRET!,
  resave: false,
  saveUninitialized: false,
  cookie: {
    httpOnly: true,
    sameSite: 'Lax',
    secure: true,   // replit preview is https
    path: '/',
  },
}));

app.post('/api/auth/login', express.json(), login);

2.	Provide admin credentials via env, not code
In .env (or Replit Secrets), set:
ADMIN_ROLE_NAME=admin
ADMIN_EMAIL=you@example.com
ADMIN_PASSWORD=your-strong-password
SESSION_SECRET=32+ char random
Make sure an admin user with that email exists (seed script or migration). Do not hardcode these in code.
