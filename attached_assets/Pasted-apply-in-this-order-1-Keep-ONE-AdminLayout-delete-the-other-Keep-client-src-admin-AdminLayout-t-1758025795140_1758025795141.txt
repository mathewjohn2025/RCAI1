4) Replace every navigate('/') in admin code
These are what Replit found causing the bounce. Replace them like this:
•	In logout handlers only → leave admin explicitly to the login page:
// client/src/components/LogoutButton.tsx
import { useNavigate } from 'react-router-dom';
import { useQueryClient } from '@tanstack/react-query';

export function LogoutButton() {
  const nav = useNavigate();
  const qc = useQueryClient();

  const onLogout = async () => {
    await fetch('/api/auth/logout', { method: 'POST', credentials: 'include' }).catch(()=>{});
    await qc.clear(); // or invalidateQueries if you prefer
    window.location.replace('/admin/login'); // hard nav to avoid SPA races
  };

  return <button onClick={onLogout}>Log out</button>;
}
